\section{Dilution $\text{C}_\text{mère}=\ldots\text{mol}\cdot \text{L}^{-1}$ vers $\text{C}_\text{fille}=\ldots\text{mol}\cdot \text{L}^{-1}$}
\vspace{0.5cm}
\noindent
\begin{minipage}[t]{0.55\textwidth}
    \textbf{Protocole} : 
    \begin{itemize}[label=--, left=0em]  % tirets plus petits et léger décalage
        \item Calcul de la dilution avec \(C_1 V_1 = C_2 V_2\).
        \item Prélèvement du volume \(V_1\) de la solution mère avec la pipette jaugée.
        \item Ajout de l'eau distillée jusqu’à $2/3$ de la fiole.
        \item Mélanger puis ajouter l'eau jusqu'au trait de jauge avec la pipette à main.
        \item Réaliser ce protocole pour toute la gamme étalon.
    \end{itemize}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.35\textwidth}
    \textbf{Matériel} : 
    \begin{itemize}[label=--, left=0em]  % tirets plus petits et léger décalage
        \item bécher
        \item pipette jaugée ($\ldots \text{mL}$)
        \item pipette à main
        \item fiole jaugée ($\ldots \text{mL}$)
        \item eau distillée
        \item solution mère 
    \end{itemize} 
\end{minipage}
